6. USAGE EXAMPLE
=============================================================================

C++ API:
--------
#include "MCP_Transport_Native.h"
#include <nlohmann/json.hpp>

// Create transport
auto transport = std::make_unique<RawrXD::MCP::Transport>(
    L"https://api.example.com"
);

// Set callbacks
transport->SetOnConnect([](const std::wstring& url) {
    std::wcout << L"Connected to: " << url << std::endl;
});

// Connect
if (transport->Connect()) {
    // Send JSON-RPC request
    nlohmann::json params = {
        {"model", "gpt-4"},
        {"prompt", "Hello, world!"}
    };
    
    auto result = transport->SendRequest("completion", params);
    if (result) {
        std::cout << "Result: " << *result << std::endl;
    }
}

C API:
------
#include "MCP_Transport_Native.h"

// Create transport
HMCPTransport hTransport = MCP_Transport_Create(
    L"https://api.example.com",
    NULL  // user data
);

// Connect
if (MCP_Transport_Connect(hTransport)) {
    // Send request
    const char* request = "{\"jsonrpc\":\"2.0\",...}";
    char* response = NULL;
    DWORD responseLen = 0;
    
    if (MCP_Transport_SendRequest(hTransport, request, strlen(request),
                                   &response, &responseLen, 30000)) {
        // Process response
        printf("Response: %.*s\n", responseLen, response);
        
        // Free response
        HeapFree(GetProcessHeap(), 0, response);
    }
}

// Cleanup
MCP_Transport_Destroy(hTransport);
