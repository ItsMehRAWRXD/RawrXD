static int ScreenPilotAuthorityRun(
    void* user,
    const RawrXD_SP_AgentRequestV2* wire,
    RawrXD_SP_EmitFnV2 emit,
    void* emit_user,
    RawrXD_SP_IsCancelledFnV2 cancelled,
    void* cancel_user,
    RawrXD_SP_RequestApprovalFnV2 approval,
    void* approval_user)
{
    auto* app = static_cast<RawrXDApp*>(user); // project-specific type

    CanonicalAgentRequest req;
    req.requestId = std::string(wire->request_id.data, wire->request_id.size);
    req.mode      = std::string(wire->mode.data, wire->mode.size);
    req.model     = std::string(wire->model.data, wire->model.size);
    req.workspace = std::string(wire->workspace.data, wire->workspace.size);
    req.prompt    = std::string(wire->prompt.data, wire->prompt.size);

    req.permissionMask = wire->permission_mask;
    req.approvalMask   = wire->approval_required_mask;
    req.workspaceOnly  = wire->workspace_only != 0;

    req.emit = [&](std::string_view type, std::string_view data) {
        return emit(emit_user, type.data(), data.data(), data.size()) != 0;
    };

    req.isCancelled = [&] {
        return cancelled(cancel_user) != 0;
    };

    req.requestApproval =
        [&](std::string_view tool,
            std::string_view summary,
            std::string_view risk,
            std::uint64_t permissionBit) {
            return approval(
                approval_user,
                std::string(tool).c_str(),
                std::string(summary).c_str(),
                std::string(risk).c_str(),
                permissionBit) != 0;
        };

    // PROJECT-SPECIFIC ADAPTER B:
    //
    // MUST be the exact same canonical entrypoint used by native IDE/CLI.
    // No CreateProcess/system/direct filesystem execution here.
    return app->CanonicalAgentCoordinator().Run(req);
}

ToolResult CanonicalToolAuthority::Invoke(
    const CanonicalAgentRequest& req,
    const ToolCall& call)
{
    using namespace rawrxd::screenpilot;

    const ToolClass cls = ClassifyTool(call.name);
    const std::uint64_t permission = PermissionBit(cls);

    // Unknown tools fail closed.
    if (cls == ToolClass::Unknown || permission == 0) {
        return Denied("unknown/unclassified tool");
    }

    if ((req.permissionMask & permission) == 0) {
        return Denied("tool denied in current ASK/PLAN/BUILD/AGENT mode");
    }

    // Enforce workspace confinement on EACH path-bearing parameter.
    for (const auto& path : ExtractFilesystemPaths(call)) {
        if (req.workspaceOnly &&
            !PathUnderWorkspace(req.workspace, path)) {
            return Denied("path outside ScreenPilot workspace");
        }
    }

    // Shell/process tools need their own command policy. Never assume cwd
    // confines a process: absolute paths, '..', shell redirection, child
    // processes, scripts and environment variables can escape it.
    if (cls == ToolClass::ProcessGeneral) {
        if (!ValidateCommandAgainstWorkspacePolicy(req, call)) {
            return Denied("process command violates workspace policy");
        }
    }

    if ((req.approvalMask & permission) != 0) {
        if (!req.requestApproval(
                call.name,
                SummarizeToolCall(call),
                RiskLabel(call),
                permission)) {
            return Denied("user denied or approval timed out");
        }
    }

    // The ONE canonical executor:
    return registry_.Invoke(call);
}
