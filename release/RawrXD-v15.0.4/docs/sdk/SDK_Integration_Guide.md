## Setting Up Your Development Environment

### Step 1: Install the SDK

```bash
# Windows
# Copy SDK files to your project
xcopy /E "C:\Program Files\SovereignIDE\SDK\*" "MyExtension\SDK\"

# Linux
# Copy SDK files to your project
cp -r /opt/sovereign-ide/sdk/* MyExtension/SDK/

# macOS
cp -r /Applications/SovereignIDE.app/Contents/SDK/* MyExtension/SDK/
```

### Step 2: Configure Your Build System

**CMake (Recommended):**

```cmake
cmake_minimum_required(VERSION 3.16)
project(MyExtension)

set(CMAKE_CXX_STANDARD 17)

# Find Sovereign SDK
find_package(SovereignSDK REQUIRED)

# Create executable
add_executable(MyExtension src/main.cpp)

# Link against SDK
target_link_libraries(MyExtension Sovereign::SDK)

# Include directories
target_include_directories(MyExtension PRIVATE ${SOVEREIGN_SDK_INCLUDE_DIR})
```

**Visual Studio:**

1. Create new C++ project
2. Add SDK include path: `Project Properties → C/C++ → General → Additional Include Directories`
3. Add SDK library path: `Project Properties → Linker → General → Additional Library Directories`
4. Link against: `sovereign_sdk.lib`

**Makefile:**

```makefile
CXX = g++
CXXFLAGS = -std=c++17 -I$(SOVEREIGN_SDK)/include
LDFLAGS = -L$(SOVEREIGN_SDK)/lib -lsovereign_sdk

MyExtension: src/main.cpp
	$(CXX) $(CXXFLAGS) -o $@ $< $(LDFLAGS)
```

## Creating Your First Extension

### Hello World Extension

```cpp
// main.cpp
#include <sovereign/sdk.h>
#include <stdio.h>

int main() {
    // Initialize SDK
    SDKConfig config = {
        .version = SDK_VERSION_1_0,
        .logLevel = LOG_LEVEL_INFO
    };
    
    SDKHandle sdk;
    SDKResult result = SDK_Initialize(&config, &sdk);
    
    if (result != SDK_SUCCESS) {
        printf("Failed to initialize SDK: %s\n", 
               SDK_GetErrorString(result));
        return 1;
    }
    
    // Get SDK version
    char version[32];
    SDK_GetVersion(sdk, version, sizeof(version));
    printf("Connected to Sovereign IDE SDK v%s\n", version);
    
    // Discover capabilities
    CapabilityInfo caps[10];
    uint32_t count;
    SDK_DiscoverCapabilities(sdk, "editor", caps, &count);
    
    printf("Found %u editor capabilities:\n", count);
    for (uint32_t i = 0; i < count; i++) {
        printf("  - %s: %s\n", caps[i].name, caps[i].description);
    }
    
    // Cleanup
    SDK_Shutdown(sdk);
    
    printf("Extension completed successfully!\n");
    return 0;
}
```

### Building and Running

```bash
# Compile
g++ -std=c++17 -I/path/to/sdk/include \
    -o MyExtension main.cpp \
    -L/path/to/sdk/lib -lsovereign_sdk

# Run
./MyExtension
```

**Expected Output:**
```

  - Editor.GetText: Get editor text content
  - Editor.SetText: Set editor text content
```

## Common Integration Patterns

### Pattern 1: Editor Extension

```cpp
// Editor extension that adds custom functionality
class EditorExtension {
private:
    SDKHandle sdk;
    EditorHandle editor;
    
public:
    bool Initialize() {
        SDKConfig config = { .version = SDK_VERSION_1_0 };
        SDKResult result = SDK_Initialize(&config, &sdk);
        return result == SDK_SUCCESS;
    }
    
    bool OpenFile(const char* path) {
        SDKResult result = SDK_Editor_OpenFile(sdk, path, &editor);
        return result == SDK_SUCCESS;
    }
    
    void InsertHeader() {
        const char* header = "// Copyright 2026\n// Generated by MyExtension\n\n";
        SDK_Editor_InsertText(sdk, editor, 0, header);
    }
    
    void Cleanup() {
        SDK_Shutdown(sdk);
    }
};

// Usage
int main() {
    EditorExtension ext;
    ext.Initialize();
    ext.OpenFile("main.cpp");
    ext.InsertHeader();
    ext.Cleanup();
    return 0;
}
```

### Pattern 2: AI-Powered Analysis

```cpp
// AI-powered code analysis extension
class AIAnalyzer {
private:
    SDKHandle sdk;
    ModelHandle model;
    
public:
    bool Initialize() {
        SDK_Initialize(NULL, &sdk);
        
        // Load analysis model
        ModelConfig config = {
            .backend = BACKEND_LLAMA_CPP,
            .contextSize = 4096
        };
        
        SDK_AI_LoadModel(sdk, "models/code-analyzer.gguf", 
                        &config, &model);
        return true;
    }
    
    AnalysisResult AnalyzeCode(const char* code) {
        // Prepare prompt
        char prompt[4096];
        snprintf(prompt, sizeof(prompt),
                 "Analyze this code for bugs and improvements:\n\n%s",
                 code);
        
        // Run inference
        InferenceInput input = {
            .prompt = prompt,
            .maxTokens = 512,
            .temperature = 0.3f
        };
        
        InferenceOutput output;
        SDK_AI_Inference(sdk, model, &input, &output);
        
        // Parse results
        AnalysisResult result;
        ParseAnalysis(output.text, &result);
        
        return result;
    }
    
    void Cleanup() {
        SDK_AI_UnloadModel(sdk, model);
        SDK_Shutdown(sdk);
    }
};
```

### Pattern 3: Binary Analysis Tool

```cpp
// Binary analysis extension
class BinaryAnalyzer {
private:
    SDKHandle sdk;
    
public:
    bool Initialize() {
        SDK_Initialize(NULL, &sdk);
        return true;
    }
    
    void AnalyzeBinary(const char* path) {
        // Load binary
        BinaryHandle binary;
        SDK_Binary_Load(sdk, path, &binary);
        
        // Get info
        BinaryInfo info;
        SDK_Binary_GetInfo(sdk, binary, &info);
        
        printf("Analyzing: %s\n", info.path);
        printf("Format: %s\n", FormatToString(info.format));
        printf("Architecture: %s\n", ArchToString(info.arch));
        
        // Get symbols
        SymbolInfo symbols[100];
        uint32_t symbolCount;
        SDK_Binary_GetSymbols(sdk, binary, symbols, &symbolCount);
        
        printf("Found %u symbols\n", symbolCount);
        
        // Disassemble main function
        if (info.entryPoint != 0) {
            Disassembly disasm;
            SDK_Disasm_Function(sdk, binary, info.entryPoint, &disasm);
            
            printf("Entry point disassembly:\n");
            for (uint32_t i = 0; i < disasm.instructionCount; i++) {
                printf("  %016llX: %s\n",
                       disasm.instructions[i].address,
                       disasm.instructions[i].text);
            }
        }
        
        // Cleanup
        SDK_Binary_Unload(sdk, binary);
    }
    
    void Cleanup() {
        SDK_Shutdown(sdk);
    }
};
```

### Pattern 4: Autonomous Agent

```cpp
// Autonomous agent extension
class AutonomousAgent {
private:
    SDKHandle sdk;
    AgentHandle agent;
    
public:
    bool Initialize() {
        SDK_Initialize(NULL, &sdk);
        
        // Create agent
        AgentConfig config = {
            .name = "CodeRefactorer",
            .type = AGENT_TYPE_AUTONOMOUS,
            .capabilities = { "refactor", "analyze", "suggest" },
            .capabilityCount = 3,
            .maxIterations = 10
        };
        
        SDK_Agent_Create(sdk, &config, &agent);
        return true;
    }
    
    void RefactorCodebase(const char* path) {
        // Execute refactoring task
        TaskOptions options = {
            .timeout = 300000,  // 5 minutes
            .maxSteps = 20,
            .allowFileOperations = true
        };
        
        TaskResult result;
        SDK_Agent_ExecuteTask(sdk, agent,
                              "Refactor this codebase to use modern C++",
                              &options, &result);
        
        if (result.success) {
            printf("Refactoring completed:\n%s\n", result.summary);
        } else {
            printf("Refactoring failed: %s\n", result.details);
        }
    }
    
    void Cleanup() {
        SDK_Agent_Destroy(sdk, agent);
        SDK_Shutdown(sdk);
    }
};
```

## Best Practices

### 1. Resource Management

```cpp
// Use RAII for automatic cleanup
class SDKResource {
private:
    SDKHandle sdk;
    bool initialized;
    
public:
    SDKResource() : initialized(false) {}
    
    ~SDKResource() {
        if (initialized) {
            SDK_Shutdown(sdk);
        }
    }
    
    bool Initialize() {
        SDKResult result = SDK_Initialize(NULL, &sdk);
        initialized = (result == SDK_SUCCESS);
        return initialized;
    }
    
    SDKHandle Get() { return sdk; }
};

// Usage
void Process() {
    SDKResource sdk;
    if (!sdk.Initialize()) {
        return;
    }
    
    // Use sdk.Get()...
    // Automatic cleanup on scope exit
}
```

### 2. Error Handling

```cpp
// Check all SDK calls
#define SDK_CHECK(call) \
    do { \
        SDKResult result = (call); \
        if (result != SDK_SUCCESS) { \
            HandleError(#call, result); \
            return false; \
        } \
    } while(0)

void HandleError(const char* operation, SDKResult result) {
    printf("SDK Error in %s: %s (code %d)\n",
           operation,
           SDK_GetErrorString(result),
           result);
}

// Usage
bool Initialize() {
    SDK_CHECK(SDK_Initialize(NULL, &sdk));
    SDK_CHECK(SDK_Editor_OpenFile(sdk, "test.cpp", &editor));
    return true;
}
```

### 3. Async Operations

```cpp
// Handle async operations properly
class AsyncOperation {
private:
    AsyncTaskHandle task;
    bool completed;
    
public:
    void Start() {
        completed = false;
        
        SDK_Agent_ExecuteTaskAsync(sdk, agent, "task",
                                   &options,
                                   OnProgress, OnComplete,
                                   this, &task);
    }
    
    static void OnProgress(uint32_t progress, uint32_t total,
                          void* userData) {
        auto* op = static_cast<AsyncOperation*>(userData);
        printf("Progress: %u/%u\n", progress, total);
    }
    
    static void OnComplete(const TaskResult* result,
                           void* userData) {
        auto* op = static_cast<AsyncOperation*>(userData);
        op->completed = true;
        
        if (result->success) {
            printf("Task completed!\n");
        } else {
            printf("Task failed: %s\n", result->error);
        }
    }
    
    void Wait() {
        while (!completed) {
            Sleep(100);
        }
    }
};
```

## Troubleshooting

### Common Issues

#### Issue 1: SDK Initialization Fails

**Symptoms:**
```
Failed to initialize SDK: SDK_ERROR_NOT_INITIALIZED
```

**Solutions:**
- Verify Sovereign IDE is running
- Check SDK version compatibility
- Ensure proper library paths

#### Issue 2: Capability Not Found

**Symptoms:**
```
SDK_ERROR_CAPABILITY_NOT_FOUND
```

**Solutions:**
- Check capability name spelling
- Verify batch is loaded
- Use SDK_DiscoverCapabilities to list available capabilities

#### Issue 3: Memory Access Violations

**Symptoms:**
```
Access violation at address 0x...
```

**Solutions:**
- Check buffer sizes
- Verify handles are valid
- Ensure proper initialization

#### Issue 4: Async Operations Hang

**Symptoms:**
- Task never completes
- Callbacks not called

**Solutions:**
- Check event loop is running
- Verify callbacks are properly registered
- Add timeout handling

### Debugging Tips

```cpp
// Enable SDK logging
SDKConfig config = {

    .logFile = "sdk_debug.log"
```

```cpp
void EnumerateCapabilitiesByBatch(SDKHandle sdk) {
```
