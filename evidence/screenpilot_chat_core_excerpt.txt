    function recordMetric(opts) {
      // opts: { latency, tokens, model, endpoint, success, errorMsg }
      var entry = {
        timestamp: Date.now(),
        latency: opts.latency || 0,
        tokens: opts.tokens || 0,
        tps: (opts.tokens > 0 && opts.latency > 0) ? parseFloat((opts.tokens / (opts.latency / 1000)).toFixed(1)) : 0,
        model: opts.model || 'unknown',
        endpoint: opts.endpoint || 'unknown',
        success: opts.success !== false,
        errorMsg: opts.errorMsg || null,
      };

      // Push to ring buffer
      State.perf.history.push(entry);
      if (State.perf.history.length > State.perf.maxHistory) {
        State.perf.history.shift();
      }

      // Update aggregates
      State.perf.totalRequests++;
      State.perf.totalTokens += entry.tokens;
      State.perf.totalLatency += entry.latency;

      // Per-model stats
      var mKey = entry.model || 'unknown';
      if (!State.perf.modelStats[mKey]) {
        State.perf.modelStats[mKey] = { requests: 0, tokens: 0, totalLatency: 0, avgTps: 0, bestTps: 0 };
      }
      var ms = State.perf.modelStats[mKey];
      ms.requests++;
      ms.tokens += entry.tokens;
      ms.totalLatency += entry.latency;
      ms.avgTps = ms.tokens > 0 && ms.totalLatency > 0 ? parseFloat((ms.tokens / (ms.totalLatency / 1000)).toFixed(1)) : 0;
      if (entry.tps > ms.bestTps) ms.bestTps = entry.tps;

      // Structured log entry
      logStructured('PERF', 'request_complete', {
        latency: entry.latency,
        tokens: entry.tokens,
        tps: entry.tps,
        model: entry.model,
        endpoint: entry.endpoint,
        success: entry.success,
      });

      // Update sidebar perf summary if visible
      updatePerfSidebar();
    }

    function logStructured(level, event, data) {
      var entry = {
        ts: Date.now(),
        isoTime: new Date().toISOString(),
        level: level,
        event: event,
        data: data || {},
      };
      State.perf.structuredLog.push(entry);
      if (State.perf.structuredLog.length > State.perf.maxLog) {
        State.perf.structuredLog.shift();
      }
    }

    /**
     * Safely parse JSON from a fetch response.
     * Handles non-JSON responses (HTML error pages, plain text) gracefully.
     * @param {Response} res - The fetch Response object
     * @param {Object} options - Options object
     * @param {string} options.endpoint - Endpoint name for logging
     * @param {boolean} options.allowEmpty - Allow empty response body
     * @returns {Promise<Object>} - Parsed JSON or error object
     */
    async function safeJsonResponse(res, options) {
      options = options || {};
      var endpoint = options.endpoint || 'unknown';
      var allowEmpty = options.allowEmpty !== false;
      
      // Check content-type before parsing JSON
      var contentType = res.headers.get('content-type') || '';
      var isJson = contentType.includes('application/json');
      
      if (!res.ok) {
        // Handle HTTP error (404, 500, etc.)
        var errorText;
        try {
          errorText = isJson ? JSON.stringify(await res.json()) : await res.text();
        } catch (e) {
          errorText = 'HTTP ' + res.status;
        }
        var errorMsg = isJson ? (errorText.error || errorText.message || 'HTTP ' + res.status) : ('HTTP ' + res.status + ' - Endpoint not found or returned HTML');
        logDebug('[' + endpoint + '] HTTP error: ' + res.status + ' - ' + (isJson ? errorText : errorText.substring(0, 200)), 'error');
        return { 
          __error: true, 
          status: res.status, 
          message: errorMsg,
          raw: errorText.substring ? errorText.substring(0, 500) : errorText
        };
      }

      // Parse JSON response safely
      if (isJson) {
        try {
          return await res.json();
        } catch (e) {
          logDebug('[' + endpoint + '] JSON parse error: ' + e.message, 'error');
          return { __error: true, message: 'Invalid JSON response: ' + e.message };
        }
      } else {
        // Not JSON content-type, try to parse anyway or return text
        var text = await res.text();
        if (!text || text.trim() === '') {
          if (allowEmpty) return {};
          return { __error: true, message: 'Empty response body' };
        }
        try {
          return JSON.parse(text);
        } catch (e) {
          logDebug('[' + endpoint + '] Non-JSON response: ' + text.substring(0, 200), 'warn');
          return { __error: true, message: 'Non-JSON response from server', raw: text.substring(0, 500) };
        }
      }
    }

    /**
     * Check if a response result is an error from safeJsonResponse
     * @param {Object} result - Result from safeJsonResponse
     * @returns {boolean} - True if result is an error
     */
    function isJsonError(result) {
      return result && result.__error === true;
    }
