void Win32NativeDebugBackend::publishState(const std::string& reason) {
    m_stopReason = reason;
    refreshVariablesUnlocked();
    // Emit S11 WFDE receipt from the WaitForDebugEvent thread itself so Native-Ext
    // does not depend solely on UI WM_APP+620 dispatch (Core TU not linked).
    if (reason.rfind("breakpoint:", 0) == 0) {
        const char* logDir = std::getenv("RAWRXD_LOG_DIR");
        const std::string path = (logDir && *logDir)
            ? (std::string(logDir) + "\\WFDE_BP_HIT_RECEIPT.txt")
            : std::string("WFDE_BP_HIT_RECEIPT.txt");
        std::ofstream out(path, std::ios::trunc);
        if (out) {
            char ripBuf[64];
            std::snprintf(ripBuf, sizeof(ripBuf), "0x%llX",
                          static_cast<unsigned long long>(m_currentRip));
            out << "WFDE_BP_HIT=1\n"
                << "STOP_REASON=" << reason << '\n'
                << "RIP=" << ripBuf << '\n'
                << "PID=" << m_pi.dwProcessId << '\n'
                << "API=WaitForDebugEvent\n";
        }
    }
    if (!m_onState) {
        return;
    }
    // Callback must not re-enter backend APIs (PostMessage / queue only).
    NativeDebugSnapshot s;
    s.state = m_state;
    s.pid = m_pi.dwProcessId;
    s.exitCode = m_exitCode;
    s.rip = m_currentRip;
    s.rsp = m_currentRsp;
    s.imageBase = m_imageBase;
    s.stopReason = m_stopReason;
    s.threads = m_threads;
    s.exports = m_exports;
    s.variables = m_variables;
    walkStackUnlocked(s.stack);
    m_onState(s);
}
