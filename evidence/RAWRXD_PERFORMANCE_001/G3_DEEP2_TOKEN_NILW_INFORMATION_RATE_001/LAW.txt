###############################################################################

FORM_A — ACTIVE_CODEC_BPW = total active representation bits / ACTIVE_WEIGHTS
  (reuse NOT yet folded into the bpw numerator)

  TOKEN_NILW
    = ACTIVE_CODEC_BPW
    × PHYSICALIZATION_RATIO
    × NEW_INFORMATION_FRACTION
    + OVERHEAD_BITS / TOTAL_WEIGHTS

  Example:
    ACTIVE_CODEC_BPW=0.8
    PHYSICALIZATION_RATIO=0.005
    NEW_INFORMATION_FRACTION=0.20
    → 0.8 × 0.005 × 0.20 = 0.0008 TOKEN_NILW

FORM_B — ACTIVE_NEW_BPW = NEW_PAYLOAD_BITS / ACTIVE_WEIGHTS
  (reuse already included in numerator)

  TOKEN_NILW
    = ACTIVE_NEW_BPW
    × PHYSICALIZATION_RATIO
    + OVERHEAD_BITS / TOTAL_WEIGHTS

  Do NOT multiply FORM_B by NEW_INFORMATION_FRACTION again.

DEFINITIONS=
  PHYSICALIZATION_RATIO = ACTIVE_LOGICAL_WEIGHTS / TOTAL_LOGICAL_WEIGHTS
  TOKEN_NILW            = NEW_PAYLOAD_BITS / TOTAL_LOGICAL_WEIGHTS
                        (+ OVERHEAD term when separately accounted)
  ACTIVE_CODEC_BPW      = representation-density metric (codec)
  ACTIVE_NEW_BPW        = new-payload density on active subset
  TOKEN_NILW            = generation-sustainability metric
