cmake_minimum_required(VERSION 3.16)
project(rawrxd_dualstick_parallel_device_partial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dsdp STATIC src/ds_parallel_device_partial.cpp)
target_include_directories(dsdp PUBLIC include)
if(MSVC)
  target_compile_options(dsdp PRIVATE /W4 /WX /permissive- /sdl)
else()
  target_compile_options(dsdp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(dsdp_selftest tests/selftest.cpp)
target_link_libraries(dsdp_selftest PRIVATE dsdp)
if(UNIX)
  target_link_libraries(dsdp_selftest PRIVATE pthread)
endif()
enable_testing()
add_test(NAME dsdp_selftest COMMAND dsdp_selftest)
