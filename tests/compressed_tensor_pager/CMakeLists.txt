cmake_minimum_required(VERSION 3.16)
project(RawrXDCompressedTensorPagerTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ZLIB REQUIRED)
enable_testing()
add_executable(compressed_tensor_pager_test test_compressed_tensor_pager.cpp)
target_include_directories(compressed_tensor_pager_test PRIVATE ../../src)
target_link_libraries(compressed_tensor_pager_test PRIVATE ZLIB::ZLIB)
if(MSVC)
 target_compile_options(compressed_tensor_pager_test PRIVATE /UNDEBUG)
else()
 target_compile_options(compressed_tensor_pager_test PRIVATE -UNDEBUG)
endif()
add_test(NAME lazy_compressed_tensor_pager COMMAND compressed_tensor_pager_test)
