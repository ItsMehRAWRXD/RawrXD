cmake_minimum_required(VERSION 3.16)
project(RawrXDCompressedPageTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ZLIB REQUIRED)
enable_testing()
add_executable(compressed_page_store_test test_compressed_page_store.cpp)
target_include_directories(compressed_page_store_test PRIVATE ../../src)
target_link_libraries(compressed_page_store_test PRIVATE ZLIB::ZLIB)
if(MSVC)
    target_compile_options(compressed_page_store_test PRIVATE /UNDEBUG)
else()
    target_compile_options(compressed_page_store_test PRIVATE -UNDEBUG)
endif()
add_test(NAME compressed_page_store COMMAND compressed_page_store_test)
