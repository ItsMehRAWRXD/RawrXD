### Loader Subdivision Logic Pseudo-Code

```powershell
# 1. Load sealed manifest
$manifest = Get-Content "manifest.json" | ConvertFrom-Json

# 2. Locate recursive snapshot artifact
$recursiveSnapshot = $manifest.artifacts | Where-Object { $_.name -eq 'recursive_sha256_snapshot.txt' }
$snapshotPath = Join-Path $distributionRoot $recursiveSnapshot.artifact
$snapshotHashes = Get-Content $snapshotPath

# 3. For each binary section to load:
foreach ($section in $manifest.sections) {
    # Lookup section path in snapshot inventory
    $sectionHash = $snapshotHashes | Select-String "path=$($section.path)"
    
    # Verify actual file hash matches snapshot entry
    $actualHash = (Get-FileHash $section.path -Algorithm SHA256).Hash
    if ($actualHash -ne $sectionHash.sha256) {
        throw "Hash mismatch for $($section.path)"
    }
    
    # Hash is valid; proceed with subdivision and VA mounting
    Mount-ToElasticVASpace -Section $section -Subdivision $section.subdivision_metadata
}
```
