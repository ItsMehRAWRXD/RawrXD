cmake_minimum_required(VERSION 3.20)
project(rawrxd_snap_supervisor LANGUAGES CXX)
add_executable(snap_supervisor snap_supervisor.cpp)
target_compile_features(snap_supervisor PRIVATE cxx_std_20)
target_compile_definitions(snap_supervisor PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(snap_supervisor PRIVATE Advapi32 Psapi)
if(MSVC)
  target_compile_options(snap_supervisor PRIVATE /W4 /EHsc)
endif()
