cmake_minimum_required(VERSION 3.20)
project(deep2_autotune LANGUAGES CXX)
add_library(deep2_autotune STATIC deep2_autotune.cpp)
target_compile_features(deep2_autotune PUBLIC cxx_std_20)
target_include_directories(deep2_autotune PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(deep2_autotune PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(NOT WIN32)
  message(FATAL_ERROR "This source drop targets native Windows")
endif()
