cmake_minimum_required(VERSION 3.20)
project(rawrxd_agent_esp_complete LANGUAGES CXX)
add_library(rawrxd_agent_esp INTERFACE)
target_include_directories(rawrxd_agent_esp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(rawrxd_agent_esp INTERFACE cxx_std_20)
add_executable(agent_esp_demo esp_demo.cpp)
target_link_libraries(agent_esp_demo PRIVATE rawrxd_agent_esp)
add_executable(agent_esp_test esp_test.cpp)
target_link_libraries(agent_esp_test PRIVATE rawrxd_agent_esp)
enable_testing()
add_test(NAME agent_esp_test COMMAND agent_esp_test)
