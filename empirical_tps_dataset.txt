# ---- DEEP2_EMPIRICAL_SCALING_AUTHORITY_002 — Five Orthogonal Core Scaling Axes ----

# RawrXD / Deep2 empirical performance authority is divided into five orthogonal
# core scaling axes. Additional measurements (latency, placement, KV growth,
# reliability, correctness, energy/resource efficiency) act as authority dimensions
# and controls, not necessarily independent scaling axes.

# AXIS_1 = MODEL_SCALING
# AXIS_2 = IMPLEMENTATION_SCALING
# AXIS_3 = CONCURRENCY_SCALING
# AXIS_4 = RESIDENCY_SCALING
# AXIS_5 = SWARM_HEX_MAG_SCALING

# ---- Axis 1 — Model Scaling ----
# What varies: exact model / model bytes
# Primary authority: exact model bytes, median measured decode TPS,
#   raw eval_count, raw eval_duration_ns
# SIZE_X = MODEL_BYTES_B / MODEL_BYTES_A
# TPS_X = DECODE_TPS_B / DECODE_TPS_A
# For positive slowdown-exponent convention:
#   TPS ∝ SIZE^-α
#   α = -ln(TPS_X) / ln(SIZE_X)
# Valid only when:
#   SIZE_X != 1
#   MODEL_BYTES_A = exact (measured file length, not nominal)
#   MODEL_BYTES_B = exact
#   same execution policy
#   same agent count
#   same workload policy
#   raw timing authority exists
# Otherwise: ALPHA=UNDEFINED
# B2/B3 belong here: same exact model, same nominal size, different engine/config
#   — B2 = direct-write implementation probe
#   — B3 = layer-split implementation experiment (KV range fixed to 32 per slot,
#     reclaims ~1.06GB VRAM/slot)
# NOT concurrency scaling (those require genuine simultaneous requests with
#   AGENT_COUNT=1,2,3 and measured wall aggregate TPS)

# ---- Axis 2 — Implementation Scaling ----
# Same: MODEL, MODEL_BYTES, AGENT_COUNT, TASK
# Changes: ENGINE, KERNEL, CACHE_POLICY, SCHEDULER, GPU_MAPPING,
#   PREFETCH_POLICY, TOKEN_OWNERSHIP_POLICY
# Examples:
#   Qwen32 baseline -> B2 direct-write implementation
#   Qwen32 baseline -> B3 layer-split implementation
# Therefore:
#   B2 != AGENT_COUNT_2
#   B3 != AGENT_COUNT_3
# They are implementation experiments, NOT concurrency rows.
# Primary metric: IMPLEMENTATION_GAIN = TPS_NEW / TPS_BASELINE

# ---- Axis 3 — Concurrency Scaling ----
# Same: MODEL, ENGINE, EXECUTION_POLICY, TASK_POLICY
# Changes: simultaneous stream count
# Required: AGENT_COUNT=1, AGENT_COUNT=2, AGENT_COUNT=3, ...
# Requests must actually overlap in wall time.
# Primary metrics:
#   WALL_AGGREGATE_TPS_N = wall aggregate TPS for N agents
#   WALL_GAIN_N = WALL_AGGREGATE_TPS_N / WALL_AGGREGATE_TPS_1
#   PER_AGENT_RETENTION_N = MEAN_AGENT_DECODE_TPS_N / SINGLE_AGENT_DECODE_TPS
# Secondary diagnostic: SUM_AGENT_DECODE_TPS
# Never substitute summed per-agent decode measurements for true wall aggregate throughput.
# Actual concurrency rows must come from genuine simultaneous requests, NOT
# implementation variation (B2/B3 are NOT concurrency rows).

# ---- Axis 4 — Residency Scaling ----
# Measures physical working set and active traffic independently of nominal model size.
# Required where available: MODEL_BYTES, PHYSICAL_VRAM_BYTES,
#   PHYSICAL_HOST_BYTES, KV_BYTES, EFFECTIVE_BYTES_TOTAL,
#   EFFECTIVE_BYTES_PER_TOKEN
# Derived:
#   RESIDENCY_RATIO = PHYSICAL_WORKING_SET_BYTES / MODEL_BYTES
#   ACTIVE_TRAFFIC_RATIO = EFFECTIVE_BYTES_PER_TOKEN / MODEL_BYTES
# Especially important for: MoE, sparse execution, expert caches,
#   selective residency, streamed weights
# A large nominal model does not prove proportionally large per-token traffic.
# That must be measured.

# ---- Axis 5 — Swarm / Hex Mag Scaling ----
# This axis measures orchestration rather than merely simultaneous inference.
# Changes can include: SWARM_SIZE, HEX_CELL_COUNT, ROLE_TOPOLOGY,
#   TASK_ROUTING, DELEGATION_POLICY, MERGE_POLICY, WORK_STEAL_POLICY,
#   MODEL_ASSIGNMENT, GPU_ASSIGNMENT
# Primary authority is not raw token count alone.
# Measure: COMPLETED_WORK_UNITS, USEFUL_WORK_UNITS, SWARM_WALL_DURATION_NS
#   SWARM_TOTAL_GENERATED_TOKENS, SWARM_USEFUL_TOKENS, SWARM_DUPLICATE_TOKENS
#   COORDINATION_WAIT_NS, HANDOFF_COUNT, MERGE_COUNT, RETRY_COUNT,
#   FAILED_BRANCH_COUNT
# Derived:
#   USEFUL_WORK_PER_SEC = USEFUL_WORK_UNITS / WALL_SECONDS
#   USEFUL_SWARM_TPS = SWARM_USEFUL_TOKENS / WALL_SECONDS
#   DUPLICATE_WORK_RATIO = DUPLICATE_WORK / TOTAL_WORK
#   SWARM_GAIN_N = USEFUL_WORK_PER_SEC_N / SINGLE_AGENT_USEFUL_WORK_PER_SEC
#   SWARM_EFFICIENCY_N = SWARM_GAIN_N / AGENT_COUNT
# This distinguishes more generated traffic from more useful autonomous engineering work.

# ---- Model-Size Alpha Computation Rule ----
# For positive slowdown-exponent convention:
#   TPS ∝ SIZE^-α
#   α = -ln(TPS_X) / ln(SIZE_X)
# Valid only when:
#   SIZE_X != 1 (distinct model bytes, not just nominal vs active)
#   MODEL_BYTES_A = exact measured file lengths, not approximate nominal values
#   MODEL_BYTES_B = exact measured file lengths
#   same execution policy
#   same agent count
#   same workload policy
#   raw timing authority exists
# If SIZE_X == 1 (same model), the result is implementation scaling, not model scaling
# If any valid condition fails: ALPHA=UNDEFINED

# Do not mix with signed slope ln(TPS_X)/ln(SIZE_X) convention.
# If you want a positive model-size slowdown exponent, use:
#   TPS ∝ SIZE^-α
#   α = -ln(TPS_B / TPS_A) / ln(SIZE_B / SIZE_A)
# rather than mixing that convention with the signed slope.

# ---- Outlier Policy ----
# Never silently discard unusual measurements.
# All raw runs remain preserved.
# Allowed RAW_RUN_STATUS values:
#   VALID — run passes all provenance and classification checks
#   FLAGGED — run has noted anomalies, still retained
#   EXCLUDED_BY_PREDECLARED_RULE — excluded only under pre-declared rule
# If excluded: EXCLUSION_REASON and EXCLUSION_RULE must be present.
# Median reporting does not authorize deletion of inconvenient results.

# ---- Multi-Dimension Runs ----
# A physical run may generate evidence for multiple dimensions.
# Example: Token Beacon Ferris run may produce:
#   IMPLEMENTATION_SCALING evidence
#   PLACEMENT_AUTHORITY evidence
#   LATENCY evidence
#   RESIDENCY evidence
#   RELIABILITY evidence
# That is valid.
# What must remain isolated is the CLAIM.
# Do not say: "TPS increased therefore Hex Mag improved"
# if both scheduler topology and hardware placement changed.
# Instead: CLAIM_VARIABLE, CONTROL_VARIABLES, CONFOUNDERS must identify
# the causal comparison.

# ---- Beacon Transfer Rule ----
# Do not transfer the complete model or complete KV cache on every token.
# Preferred structure: persistent model/state replica + per-token mutable delta
# + control beacon
# Conceptually: GPU A computes token N --> token ID + KV append + mutable state
# delta --> BEACON --> GPU B computes token N+1
# For hardware without direct peer access:
#   GPU A --> GPU DMA --> pinned host-visible staging --> GPU DMA --> GPU B

# ---- Token-State Coherency ----
# Every mutable state dependency between token turns must be identified.
# At minimum: TOKEN_ID, POSITION, ROPE_POSITION, KV_APPEND, KV_EPOCH,
# SAMPLER_RNG_STATE, SAMPLING_HISTORY, PENALTY_STATE, GRAMMAR_STATE,
# SPECULATION_STATE, MOE_ROUTING_STATE, EXPERT_CACHE_GENERATION,
# WINDOW_STATE, ATTENTION_SINK_STATE, CACHE_EVICTION_GENERATION
# A token flip is not valid until the next hardware owner has authoritative
# state for every required mutable dependency.

# ---- Beacon Authority Fields ----
# Record: TOKEN_INDEX, TOKEN_OWNER, BEACON_SEQUENCE, BEACON_BYTES,
# BEACON_CONTROL_NS, KV_DELTA_BYTES, KV_TRANSFER_NS,
# FENCE_WAIT_NS, QUEUE_IDLE_NS, TOKEN_COMPUTE_NS,
# NEXT_DEVICE_PREP_NS, INTER_TOKEN_NS,
# GPU0_BUSY_NS, GPU1_BUSY_NS,
# GPU0_PREFETCH_OVERLAP_NS, GPU1_PREFETCH_OVERLAP_NS
# Derived:
#   BEACON_OVERHEAD_RATIO = (BEACON_CONTROL_NS + KV_TRANSFER_NS + FENCE_WAIT_NS) /
#     WALL_DECODE_NS
#   PREPARATION_HIDDEN_RATIO = STANDBY_PREPARATION_OVERLAPPED_NS /

# Core measurement formulas (no assumptions, all values from measured receipts)
# TPS = (tokens × 10^9) / decode_ns
# SIZE_X = MODEL_BYTES / BASE_MODEL_BYTES
# TPS_X = MEASURED_TPS / BASE_MEASURED_TPS
# TOKEN_COST_X = NS_PER_TOKEN / BASE_NS_PER_TOKEN

# Reverse-engineered exponents
# α = -ln(TPS2/TPS1) / ln(Size2/Size1) — how TPS scales with model size
#   α = 1.0: linear decline; α < 1.0: slower decline; α > 1.0: faster decline
#   Only computed when SIZE_X != 1 (distinct model sizes)
# β = ln(Cost2/Cost1) / ln(Size2/Size1) — measured token cost scaling

# Key principle: NO extrapolation

# MODEL_SIZE_X can rise dramatically while BYTES_PER_TOKEN_X rises much less
