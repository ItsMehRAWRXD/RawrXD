cmake_minimum_required(VERSION 3.20)
project(RawrXD_IDE_Completion_006 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rawr_ide_gate_core STATIC src/rawr_ide_gates.cpp)
target_include_directories(rawr_ide_gate_core PUBLIC include)
add_executable(rawr-ide-gates src/main.cpp)
target_link_libraries(rawr-ide-gates PRIVATE rawr_ide_gate_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(rawr-ide-gates-tests tests/test_gates.cpp)
 target_link_libraries(rawr-ide-gates-tests PRIVATE rawr_ide_gate_core)
 add_test(NAME ide_gate_contract COMMAND rawr-ide-gates-tests)
endif()
