Ollama's current source builds blob paths from the active model root as `blobs\sha256-...`, with the digest obtained from the manifest; `ollama cp` copies the manifest rather than exporting a new GGUF. :chatgpt-content-reference{index="0"}

Run this directly:

```powershell
$roots = @(
    $env:OLLAMA_MODELS,
    [Environment]::GetEnvironmentVariable('OLLAMA_MODELS','User'),
    [Environment]::GetEnvironmentVariable('OLLAMA_MODELS','Machine'),
    "$HOME\.ollama\models",
    "G:\OllamaModels"
) | Where-Object { $_ } | Select-Object -Unique

foreach ($root in $roots) {
    Write-Host "`n=== ROOT: $root ===" -ForegroundColor Cyan

    $manifest = Join-Path $root "manifests\registry.ollama.ai\library\nemotron-3.5-lightning\30b"

    if (!(Test-Path $manifest)) {
        Write-Host "Manifest not found"
        continue
    }

    Write-Host "MANIFEST=$manifest" -ForegroundColor Green

    $m = Get-Content $manifest -Raw | ConvertFrom-Json

    $m.layers | ForEach-Object {
        $blob = Join-Path $root "blobs\$($_.digest -replace ':','-')"

        [PSCustomObject]@{
            MediaType = $_.mediaType
            Digest    = $_.digest
            SizeGB    = [math]::Round($_.size / 1GB, 3)
            Exists    = Test-Path $blob
            Blob      = $blob
        }
    } | Format-Table -AutoSize
}
```

### Prove it's GGUF

Do this before Deep2:

```powershell
$fs = [IO.File]::OpenRead($blob)
$b = New-Object byte[] 4
[void]$fs.Read($b,0,4)
$fs.Close()

$magic = [BitConverter]::ToString($b)

Write-Host "MAGIC=$magic"

if ($magic -eq "47-47-55-46") {
    Write-Host "GGUF=PASS" -ForegroundColor Green
} else {
    Write-Host "GGUF=FAIL" -ForegroundColor Red
}
```

Expected:

```text
MAGIC=47-47-55-46
GGUF=PASS
```

Ollama's code explicitly maps manifest digests such as `sha256:abcd...` to files named `sha256-abcd...` under the model root's `blobs` directory. :chatgpt-content-reference{index="1"}
