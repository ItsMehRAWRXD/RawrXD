cmake_minimum_required(VERSION 3.20)
project(rawrxd_ide_gates LANGUAGES CXX)
include(CTest)
add_executable(rawrxd-ide-gates src/rawr_ide_gates.cpp tools_main.cpp)
target_include_directories(rawrxd-ide-gates PRIVATE include)
target_compile_features(rawrxd-ide-gates PRIVATE cxx_std_20)
if(BUILD_TESTING)
 add_executable(rawrxd-ide-gates-test src/rawr_ide_gates.cpp tests/test_gates.cpp)
 target_include_directories(rawrxd-ide-gates-test PRIVATE include)
 target_compile_features(rawrxd-ide-gates-test PRIVATE cxx_std_20)
 add_test(NAME rawrxd_ide_gates COMMAND rawrxd-ide-gates-test)
endif()
