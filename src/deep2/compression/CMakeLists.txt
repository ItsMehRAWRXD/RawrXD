cmake_minimum_required(VERSION 3.20)
add_library(rawrxd_compressed_weight_store CompressedWeightStore.cpp)
target_include_directories(rawrxd_compressed_weight_store PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(rawrxd_compressed_weight_store PUBLIC cxx_std_20)
if(MSVC)
 target_compile_options(rawrxd_compressed_weight_store PRIVATE /W4 /permissive-)
else()
 target_compile_options(rawrxd_compressed_weight_store PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(test_compressed_weight_store ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/compressed_weight_store/test_compressed_weight_store.cpp)
 target_link_libraries(test_compressed_weight_store PRIVATE rawrxd_compressed_weight_store)
 add_test(NAME compressed_weight_store_roundtrip COMMAND test_compressed_weight_store)
endif()
