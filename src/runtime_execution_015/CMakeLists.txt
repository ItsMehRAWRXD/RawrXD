cmake_minimum_required(VERSION 3.20)
project(RawrXDRuntime015 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(rawr-runtime src/main.cpp)
add_executable(rawr-runtime-tests tests/test_http.cpp)
target_include_directories(rawr-runtime PRIVATE src)
target_include_directories(rawr-runtime-tests PRIVATE src)
if(WIN32)
 target_link_libraries(rawr-runtime PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME runtime_parser_negative_cases COMMAND rawr-runtime-tests)
add_test(NAME runtime_fail_closed COMMAND rawr-runtime probe 127.0.0.1 1 ${CMAKE_CURRENT_BINARY_DIR}/failed.json)
set_tests_properties(runtime_fail_closed PROPERTIES WILL_FAIL TRUE)
