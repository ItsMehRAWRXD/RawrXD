cmake_minimum_required(VERSION 3.15)
project(UniversalAsyncGGUFRunner LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(UniversalRunner UniversalRunner.cpp)

# Enable deep hardware optimization flags dynamically based on active toolchain mappings
if(MSVC)
    # Check if the host compiler version explicitly supports 512-bit vector processing arguments
    if(MSVC_VERSION GREATER_EQUAL 1910)
        target_compile_options(UniversalRunner PRIVATE 
            /O2 /Oi /Ot /EHsc /W4 /permissive-
            /arch:AVX512 # Enforces native AVX-512 instruction generation inside vector lanes
        )
        message(STATUS "[BUILD] MSVC compiler target assigned to strict high-density AVX-512 instructions.")
    else()
        target_compile_options(UniversalRunner PRIVATE /O2 /Oi /Ot /EHsc /W4 /arch:AVX2)
        message(STATUS "[BUILD] Standard AVX2 micro-architectural fallbacks configured.")
    endif()
else()
    target_compile_options(UniversalRunner PRIVATE 
        -O3 -Wall -Wextra -ffast-math -pthread
    )
    include(CheckCXXCompilerFlag)
    CHECK_CXX_COMPILER_FLAG("-mavx512f" HAS_AVX512)
    if(HAS_AVX512)
        target_compile_options(UniversalRunner PRIVATE -mavx512f -mavx512bw -mavx512dq -mavx512vnni)
    endif()
endif()

target_include_directories(UniversalRunner PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
