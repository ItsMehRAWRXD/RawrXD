# Scalar Agentic IDE Build Configuration

cmake_minimum_required(VERSION 3.15)

# Add agentic IDE executable
add_executable(RawrXD-Agentic-IDE
    src/agentic_ide_main.cpp
    src/agentic_ide.cpp
    src/scalar_server.cpp
    src/file_browser.cpp
    src/chat_interface.cpp
    src/agentic_engine.cpp
    src/inference_engine.cpp
    src/gguf_loader.cpp
    src/transformer_block_scalar.cpp
    src/vulkan_compute.cpp
    src/settings.cpp
)

target_include_directories(RawrXD-Agentic-IDE PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include/backend
)

# Link with WinSock for server functionality
if(WIN32)
    target_link_libraries(RawrXD-Agentic-IDE ws2_32)
endif()

# Set C++17 standard
target_compile_features(RawrXD-Agentic-IDE PRIVATE cxx_std_17)

# Installation
install(TARGETS RawrXD-Agentic-IDE
    RUNTIME DESTINATION bin
)

message(STATUS "Configured Scalar Agentic IDE build")
message(STATUS "  - Features: Integrated server, file browser, chat, autonomous agent")
message(STATUS "  - Mode: 100% Scalar (no threading, no GPU, no SIMD)")
message(STATUS "  - Server: Embedded HTTP/WebSocket on port 8080")
