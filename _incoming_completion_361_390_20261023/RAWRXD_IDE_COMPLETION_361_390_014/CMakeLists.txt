cmake_minimum_required(VERSION 3.20)
project(RawrXDCompletion014 LANGUAGES CXX)
add_executable(rawr-runtime-gates src/main.cpp)
target_compile_features(rawr-runtime-gates PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(rawr-runtime-gates PRIVATE /W4 /EHsc)
else()
 target_compile_options(rawr-runtime-gates PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME partial_must_fail COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:rawr-runtime-gates> -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -DOUT=${CMAKE_CURRENT_BINARY_DIR} -DCASE=partial -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check.cmake)
add_test(NAME positive_fixture COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:rawr-runtime-gates> -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -DOUT=${CMAKE_CURRENT_BINARY_DIR} -DCASE=positive -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check.cmake)
add_test(NAME malformed_rejected COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:rawr-runtime-gates> -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -DOUT=${CMAKE_CURRENT_BINARY_DIR} -DCASE=malformed -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check.cmake)
