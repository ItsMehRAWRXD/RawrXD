cmake_minimum_required(VERSION 3.20)
project(deep2_tiered_moe LANGUAGES CXX)
add_library(deep2_tiered_moe STATIC src/deep2_tiered_moe.cpp)
target_include_directories(deep2_tiered_moe PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(deep2_tiered_moe PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(deep2_tiered_moe PRIVATE /W4 /EHsc)
else()
  target_compile_options(deep2_tiered_moe PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(deep2_tiered_moe_test tests/tiered_moe_test.cpp)
  target_link_libraries(deep2_tiered_moe_test PRIVATE deep2_tiered_moe)
  add_test(NAME deep2_tiered_moe_test COMMAND deep2_tiered_moe_test)
endif()
