cmake_minimum_required(VERSION 3.20)
project(rguf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rguf STATIC RGUFPlatform.cpp RGUFQuant.cpp RGUFWriter.cpp RGUFLoader.cpp)
target_include_directories(rguf PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_link_libraries(rguf PUBLIC bcrypt)
endif()
add_executable(rguf_tool rguf_tool.cpp)
target_link_libraries(rguf_tool PRIVATE rguf)
